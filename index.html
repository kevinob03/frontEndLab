JavaScript añade comportamiento

Un evento modifica el DOM sin recargar la página ni enviar información.

javascriptDemo.js

JS
const formulario = document.querySelector('#registro');
const mensaje = document.querySelector('#mensaje');

function mostrarConfirmacion(correo) {
  mensaje.textContent = `Correo válido: ${correo}`;
}

formulario.addEventListener('submit', (evento) => {
  evento.preventDefault();
  if (!formulario.checkValidity()) {
    formulario.reportValidity();
    return;
  }
  mostrarConfirmacion(formulario.elements.correo.value);
});

Una API conecta aplicaciones

Fetch realiza una petición GET real y convierte la respuesta a JSON.

apiDemo.js

API
async function consultarUsuario() {
  try {
    const response = await fetch(API_URL);
    estado.textContent = response.status;

    if (!response.ok) {
      throw new Error('Respuesta no válida');
    }

    const usuario = await response.json();
    nombre.textContent = usuario.name;
  } catch (error) {
    mostrarError(error.message);
  }
}
